(resources.Get .) -}} {{- $data = .Content -}} {{- else -}} {{- $_resources := $.Page.Resources.ByType "application" -}} {{- $_resources = (where $_resources "MediaType.SubType" "yaml") -}} {{- if $_resources -}} {{- errorf "file not found: %q, did you mean one of: %s" . $_resources -}} {{- else -}} {{- errorf "file not found: %q" . -}} {{- end -}} {{- end -}} {{- else -}} {{- errorf "no 'file' parameter, only %v" $_keys -}} {{- end -}} {{- $contents := slice -}} {{- with $data -}} {{- with try (transform.Unmarshal (dict "format" "yaml") $data) -}} {{- with .Value -}} {{- $contents = . -}} {{- else -}} {{- errorf "could not parse %q: %v" $filename .Err -}} {{- end -}} {{- end -}} {{- end -}} {{- $id := printf "%s-%s-%s" (xxhash $filename) (xxhash $data) (xxhash .Inner) -}} {{- $template := resources.FromString (printf "datatable-%s" $id) .Inner -}} {{- $_heads := 0 -}} {{- $headers := slice -}} {{- with .Get "headers" -}} {{- $_heads = add $_heads 1 -}} {{- $headers = split .
