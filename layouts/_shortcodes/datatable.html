"," -}} {{- else -}} {{- errorf "no 'headers' parameter, only %v" $_keys -}} {{- end -}} {{- $data := "" -}} {{- $filename := "" -}} {{- with .Get "file" -}} {{- $filename = . -}} {{- with or ($.Page.Resources.Get .) (resources.Get .) -}} {{- $data = .Content -}} {{- else -}} {{- $_resources := $.Page.Resources.ByType "application" -}} {{- $_resources = (where $_resources "MediaType.SubType" "yaml") -}} {{- if $_resources -}} {{- errorf "file not found: %q, did you mean one of: %s" . $_resources -}} {{- else -}} {{- errorf "file not found: %q" . -}} {{- end -}} {{- end -}} {{- else -}} {{- errorf "no 'file' parameter, only %v" $_keys -}} {{- end -}} {{- $contents := slice -}} {{- with $data -}} {{- with try (transform.Unmarshal (dict "format" "yaml") $data) -}} {{- with .Value -}} {{- $contents = . -}} {{- else -}} {{- errorf "could not parse %q: %v" $filename .Err -}} {{- end -}} {{- end -}} {{- end -}} {{- $id := printf "%s-%s-%s" (xxhash $filename) (xxhash $data) (xxhash .Inner) -}} {{- $template := resources.FromString (printf "datatable-%s" $id) .Inner -}} {{- range $headers -}} {{ . }} {{- end -}} {{- range $idx, $el := $contents -}} {{- $tr := resources.ExecuteAsTemplate (printf "datatable-%s-%d.html" $id $idx) $el $template -}} {{- with try $tr.Content -}} {{- with .Value -}} {{- . | safeHTML -}} {{- else -}} {{ errorf "invalid template for %s shortcode at %s %v" $.Name $.Position .Err }} {{- end -}} {{- end -}} {{- end -}}
